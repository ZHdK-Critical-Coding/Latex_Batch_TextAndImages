% One PDF with all folders in source/. Compile with LuaLaTeX:
%   lualatex -jobname=document -output-directory=output template.tex
% build.sh / build.ps1 do exactly this.
\documentclass[a4paper,11pt]{article}
\usepackage[margin=20mm]{geometry}
\usepackage{fontspec}   % Unicode text (äöü …); default font: Latin Modern
\usepackage{graphicx}   % \includegraphics
\pagestyle{empty}       % no page numbers
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.6em}
\frenchspacing           % one normal space after a full stop

% the title of a folder (first line of text.txt)
\newcommand{\itemtitle}[1]{{\Huge\bfseries #1\par}\vspace{1em}}

% one image on its own page, fitted into the margins and centred
\newcommand{\itemimage}[1]{%
  \clearpage\null\vfill
  \begin{center}
    \includegraphics[width=\textwidth,height=\textheight,keepaspectratio]{#1}
  \end{center}
  \vfill}

% batch.lua reads source/ and calls \itemtitle and \itemimage
\directlua{batch = require("batch")}

\begin{document}

% for every folder: a new page with title and text, then one page per image
\directlua{batch.items()}

\end{document}
