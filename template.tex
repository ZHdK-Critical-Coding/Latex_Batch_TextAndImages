% One PDF with all folders in source/. build.sh / build.ps1 write the
% content into output/content.tex and then compile this file with XeLaTeX:
%   xelatex -jobname=document -output-directory=output template.tex
\documentclass{article}
\usepackage[a5paper,margin=12mm]{geometry}   % A5, 12 mm page border
\usepackage{fontspec}   % system fonts and Unicode text (äöü …)
\usepackage[ngerman]{babel}   % German hyphenation
\usepackage{graphicx}   % \includegraphics
\usepackage{adjustbox}  % scale images to cover the page
\usepackage{eso-pic}    % place images on the page, ignoring the margins
\pagestyle{empty}       % no page numbers
\frenchspacing           % one normal space after a full stop

% Helvetica 14/16 pt, regular only. Lines are 16 pt apart, the first
% baseline is 16 pt below the top border, paragraphs are separated by
% one empty line (16 pt). Text is justified (the LaTeX default).
\setmainfont{Helvetica}
\AtBeginDocument{\fontsize{14pt}{16pt}\selectfont}
\setlength{\topskip}{16pt}
\setlength{\parindent}{0pt}
\setlength{\parskip}{16pt}

% one image on its own page, full bleed: scaled to cover the whole page,
% centred, whatever sticks out over the page edge is cut off
% (\detokenize allows & and other special characters in the file name)
\newcommand{\itemimage}[1]{%
  \clearpage
  \AddToShipoutPictureBG*{\AtPageCenter{\makebox[0pt]{\raisebox{-0.5\height}{%
    \adjustbox{width=\paperwidth, min height=\paperheight}
      {\includegraphics{\detokenize{#1}}}}}}}%
  \null\clearpage}

\begin{document}

% for every folder: a new page with the text, then one page per image
\input{output/content.tex}

\end{document}
